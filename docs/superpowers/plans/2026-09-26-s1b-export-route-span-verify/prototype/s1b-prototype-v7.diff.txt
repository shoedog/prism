+//! S1b: what a JS/TS identifier statically denotes. The nearest scope that declares the
+//! name decides. The binding denotes one callable only when that scope holds exactly one
+//! declaration of it, the declaration's value is a function (or an admitted React wrapper
+//! of one), the scope parses cleanly, and no write in the scope targets it. Every other
+//! outcome is a refusal. Runtime mutation is out of model (CLAUDE.md, static-binding Exact).
+use super::{is_js_ts_function_like, ParsedFile};
+use std::collections::BTreeSet;
+use tree_sitter::Node;
+
+/// A proven callable: its line span, and whether it is a wrapped React render function
+/// (bindable from JSX element sites only).
+#[derive(
+    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, serde::Serialize, serde::Deserialize,
+)]
+pub struct JsTerminal {
+    pub start_line: usize,
+    pub end_line: usize,
+    pub wrapped: bool,
+}
+
+/// Per-file memo: `(scope node id, name)` -> the binding's terminal, or `None` when that
+/// scope does not declare the name.
+pub type JsBindingCache =
+    std::collections::BTreeMap<(usize, String), Option<Result<JsTerminal, &'static str>>>;
+
+fn is_scope(kind: &str, parent: Option<&str>) -> bool {
+    match kind {
+        "program"
+        | "for_statement"
+        | "for_in_statement"
+        | "catch_clause"
+        | "switch_body"
+        | "class_declaration"
+        | "class"
+        | "abstract_class_declaration"
+        | "with_statement"
+        | "formal_parameters" => true,
+        "statement_block" => !parent.is_some_and(is_js_ts_function_like),
+        k => is_js_ts_function_like(k),
+    }
+}
+
+/// A parameter list sees its own function's parameters, never the body's declarations,
+/// so the walk continues above that function.
+fn next_scope_up(scope: Node<'_>) -> Option<Node<'_>> {
+    match scope.kind() {
+        "formal_parameters" => scope.parent().and_then(|f| f.parent()),
+        _ => scope.parent(),
+    }
+}

+impl ParsedFile {
+    /// The nearest scope enclosing `site` that declares `name`, with its declarations.
+    pub(crate) fn js_ts_visible_binding<'a>(
+        &self,
+        site: Node<'a>,
+        name: &str,
+    ) -> Option<(Node<'a>, Vec<Node<'a>>)> {
+        let mut current = site.parent();
+        while let Some(scope) = current {
+            if is_scope(scope.kind(), scope.parent().map(|p| p.kind())) {
+                let decls = self.js_ts_scope_declarations(scope, name);
+                if !decls.is_empty() {
+                    return Some((scope, decls));
+                }
+            }
+            current = next_scope_up(scope);
+        }
+        None
+    }

+    /// The callable `name` denotes at `site`, memoized per `(scope, name)` for one file.
+    pub(crate) fn js_ts_site_terminal(
+        &self,
+        site: Node<'_>,
+        name: &str,
+        cache: &mut JsBindingCache,
+    ) -> Result<JsTerminal, &'static str> {
+        let mut current = site.parent();
+        while let Some(scope) = current {
+            if is_scope(scope.kind(), scope.parent().map(|p| p.kind())) {
+                let key = (scope.id(), name.to_string());
+                if !cache.contains_key(&key) {
+                    let decls = self.js_ts_scope_declarations(scope, name);
+                    let found = (!decls.is_empty()).then(|| {
+                        self.js_ts_binding_terminal(scope, &decls, name)
+                            .map(|t| t.1)
+                    });
+                    cache.insert(key.clone(), found);
+                }
+                if let Some(result) = cache[&key] {
+                    return result;
+                }
+            }
+            current = next_scope_up(scope);
+        }
+        Err("unbound")
+    }

+    /// Every node that declares `name` in `scope` itself (not in nested scopes, except
+    /// `var` hoisted to a function or the program). A `with` statement declares any name.
+    pub(crate) fn js_ts_scope_declarations<'a>(
+        &self,
+        scope: Node<'a>,
+        name: &str,
+    ) -> Vec<Node<'a>> {
+        let mut out = Vec::new();
+        let named = |n: Option<Node<'_>>| n.is_some_and(|n| self.node_text(&n) == name);
+        match scope.kind() {
+            "with_statement" => out.push(scope),
+            "catch_clause" | "for_in_statement" => {
+                let field = if scope.kind() == "catch_clause" {
+                    "parameter"
+                } else {
+                    "left"
+                };
+                let lexical = scope.kind() == "catch_clause"
+                    || scope
+                        .child_by_field_name("kind")
+                        .is_some_and(|k| matches!(self.node_text(&k), "let" | "const"));
+                if lexical && self.pattern_binds(scope.child_by_field_name(field), name) {
+                    out.push(scope);
+                }
+            }
+            "for_statement" => {
+                if let Some(init) = scope.child_by_field_name("initializer") {
+                    if init.kind() == "lexical_declaration" {
+                        self.scope_walk(init, name, true, false, &mut out);
+                    }
+                }
+            }
+            "class_declaration" | "class" | "abstract_class_declaration" => {
+                if named(scope.child_by_field_name("name")) {
+                    out.push(scope);
+                }
+            }
+            "formal_parameters" => {
+                let f = scope.parent().filter(|f| is_js_ts_function_like(f.kind()));
+                let mut params = BTreeSet::new();
+                if let Some(f) = f {
+                    self.collect_js_ts_parameter_bindings(f, &mut params);
+                }
+                if params.contains(name) {
+                    out.push(scope);
+                } else if f.is_some_and(|f| {
+                    matches!(f.kind(), "function_expression" | "generator_function")
+                        && named(f.child_by_field_name("name"))
+                }) {
+                    out.extend(f);
+                }
+            }
+            k if is_js_ts_function_like(k) => {
+                let mut params = BTreeSet::new();
+                self.collect_js_ts_parameter_bindings(scope, &mut params);
+                // Every non-arrow function implicitly binds `arguments` (not a callable).
+                if params.contains(name)
+                    || (name == "arguments" && k != "arrow_function")
+                    || self.pattern_binds(scope.child_by_field_name("parameter"), name)
+                {
+                    out.push(scope.child_by_field_name("parameters").unwrap_or(scope));
+                } else if matches!(k, "function_expression" | "generator_function")
+                    && named(scope.child_by_field_name("name"))
+                {
+                    out.push(scope);
+                }
+                if let Some(body) = scope.child_by_field_name("body") {
+                    if body.kind() == "statement_block" {
+                        self.scope_walk(body, name, true, true, &mut out);
+                    }
+                }
+            }
+            "program" => self.scope_walk(scope, name, true, true, &mut out),
+            _ => self.scope_walk(scope, name, true, false, &mut out),
+        }
+        out
+    }
+
+    fn pattern_binds(&self, pattern: Option<Node<'_>>, name: &str) -> bool {
+        let mut names = BTreeSet::new();
+        if let Some(p) = pattern {
+            self.collect_js_ts_binding_pattern_names(p, &mut names);
+        }
+        names.contains(name)
+    }

+    /// Declarations under `node`: lexical ones only while `direct` (still in the scope's own
+    /// statement list), `var` ones anywhere while `hoist`. Stops at nested functions and
+    /// classes, whose own names bind here only when direct.
+    fn scope_walk<'a>(
+        &self,
+        node: Node<'a>,
+        name: &str,
+        direct: bool,
+        hoist: bool,
+        out: &mut Vec<Node<'a>>,
+    ) {
+        let mut cursor = node.walk();
+        for ch in node.named_children(&mut cursor) {
+            let kind = ch.kind();
+            let own_name = ch
+                .child_by_field_name("name")
+                .is_some_and(|n| self.node_text(&n) == name);
+            if is_js_ts_function_like(kind) || kind.contains("class") {
+                let declares = matches!(
+                    kind,
+                    "function_declaration"
+                        | "generator_function_declaration"
+                        | "class_declaration"
+                        | "abstract_class_declaration"
+                );
+                if direct && declares && own_name {
+                    out.push(ch);
+                } else if hoist && own_name && kind.ends_with("function_declaration") {
+                    // Annex B.3.3: a block function is also var-hoisted in sloppy code. The
+                    // marker (its block) is not a callable, so outside the block it refuses.
+                    out.push(node);
+                }
+                continue;
+            }
+            match kind {
+                "variable_declarator" => {
+                    if self.pattern_binds(ch.child_by_field_name("name"), name) {
+                        out.push(ch);
+                    }
+                }
+                "variable_declaration" if !hoist => continue,
+                "lexical_declaration" if !direct => continue,
+                "for_in_statement"
+                    if hoist
+                        && ch
+                            .child_by_field_name("kind")
+                            .is_some_and(|k| self.node_text(&k) == "var")
+                        && self.pattern_binds(ch.child_by_field_name("left"), name) =>
+                {
+                    out.push(ch);
+                }
+                "import_statement" => {
+                    if direct && self.js_ts_import_binds(ch, name) {
+                        out.push(ch);
+                    }
+                    continue;
+                }
+                "ambient_declaration"
+                | "enum_declaration"
+                | "internal_module"
+                | "module"
+                | "import_alias"
+                | "interface_declaration"
+                | "type_alias_declaration" => {
+                    if direct && (own_name || self.ambient_binds(ch, name)) {
+                        out.push(ch);
+                    }
+                    continue;
+                }
+                _ => {}
+            }
+            let transparent = matches!(
+                kind,
+                "export_statement"
+                    | "expression_statement"
+                    | "switch_case"
+                    | "switch_default"
+                    | "lexical_declaration"
+                    | "variable_declaration"
+            );
+            self.scope_walk(ch, name, direct && transparent, hoist, out);
+        }
+    }
+
+    fn ambient_binds(&self, node: Node<'_>, name: &str) -> bool {
+        if node.kind() != "ambient_declaration" {
+            return false;
+        }
+        let mut found = Vec::new();
+        self.scope_walk(node, name, true, true, &mut found);
+        let mut cursor = node.walk();
+        let sig = node.named_children(&mut cursor).any(|n| {
+            n.kind() == "function_signature"
+                && n.child_by_field_name("name")
+                    .is_some_and(|x| self.node_text(&x) == name)
+        });
+        sig || !found.is_empty()
+    }
+
+    fn js_ts_import_binds(&self, node: Node<'_>, name: &str) -> bool {
+        let mut cursor = node.walk();
+        let found = node.named_children(&mut cursor).any(|n| match n.kind() {
+            "identifier" => self.node_text(&n) == name,
+            "import_specifier" => n
+                .child_by_field_name("alias")
+                .or_else(|| n.child_by_field_name("name"))
+                .is_some_and(|x| self.node_text(&x) == name),
+            "string" => false,
+            _ => self.js_ts_import_binds(n, name),
+        });
+        found
+    }

+    /// The callable a binding holds, or why it holds none (SPEC §3.1).
+    pub(crate) fn js_ts_binding_terminal(
+        &self,
+        scope: Node<'_>,
+        decls: &[Node<'_>],
+        name: &str,
+    ) -> Result<(String, JsTerminal), &'static str> {
+        if scope.has_error() {
+            return Err("parse_recovery");
+        }
+        let [decl] = decls else {
+            return Err("duplicate_declaration");
+        };
+        let plain = |n: Node<'_>| {
+            let (start_line, end_line) = self.node_line_range(&n);
+            let local = self
+                .language
+                .function_name(&n)
+                .map(|x| self.node_text(&x).to_string());
+            (
+                local.unwrap_or_default(),
+                JsTerminal {
+                    start_line,
+                    end_line,
+                    wrapped: false,
+                },
+            )
+        };
+        let terminal = match decl.kind() {
+            "function_declaration" | "generator_function_declaration" => plain(*decl),
+            "function_expression" if decl.id() == scope.id() => plain(*decl),
+            "variable_declarator"
+                if decl
+                    .child_by_field_name("name")
+                    .is_some_and(|n| n.kind() == "identifier") =>
+            {
+                let value = decl.child_by_field_name("value").ok_or("not_callable")?;
+                match value.kind() {
+                    "arrow_function" | "function_expression" => plain(value),
+                    "call_expression" => {
+                        let parent = decl.parent().ok_or("not_callable")?;
+                        let (local, s, e) = self
+                            .js_ts_wrapped_export(parent, *decl)
+                            .map_err(|_| "call_not_admitted")?;
+                        let wrapped = !self.js_ts_passthrough_wrapper(value);
+                        (
+                            local,
+                            JsTerminal {
+                                start_line: s,
+                                end_line: e,
+                                wrapped,
+                            },
+                        )
+                    }
+                    _ => return Err("not_callable"),
+                }
+            }
+            _ => return Err("not_callable"),
+        };
+        // MEASUREMENT ONLY: at module scope, the base write scan instead of the scoped one.
+        let written = if scope.kind() == "program"
+            && std::env::var_os("PRISM_S1B_BASE_WRITE_SCAN").is_some()
+        {
+            self.js_ts_module_value_written(name)
+        } else {
+            self.js_ts_scope_value_written(scope, name)
+        };
+        if written {
+            return Err("written");
+        }
+        Ok(terminal)
+    }

+            return Err("undeclared");
+        }
+        self.js_ts_binding_terminal(root, &decls, name)
+    }

+        name: String,
+        facts: &mut crate::js_exports::JsExportFacts,

+            }
+        }

+    /// A write to the binding of `name` owned by `scope`.
