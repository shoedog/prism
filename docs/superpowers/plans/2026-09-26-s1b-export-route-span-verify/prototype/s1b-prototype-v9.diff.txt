+//! S1b: what a JS/TS name statically denotes (SPEC §3.1). The nearest scope that declares
+//! the name decides; the binding denotes one callable only when that scope holds exactly one
+//! declaration, the declaration's value is a function (or an admitted React wrapper of one),
+//! the scope parses cleanly (or its errors are sealed), and every node kind in the scope is
+//! classified. A written binding, or a declarator whose value is any other call, is the
+//! may-call class and keeps base behavior (owner E5). Runtime mutation is out of model.

+/// render function (bindable from JSX element sites only).
+#[derive(

+pub struct JsTerminal {
+    pub local: String,
+    pub start_line: usize,
+    pub end_line: usize,
+    pub wrapped: bool,

+/// What a name denotes where it is used.

+pub enum JsBinding {

+    MayCall,
+    Refused(&'static str),
+    /// Owner OQ1 (K): an unproven position keeps base behavior, counted by reason.
+    Base(&'static str),

+/// Σ (SPEC §3.1): the node kinds that open an environment.
+pub(super) fn is_scope(node: Node<'_>) -> bool {
+    match node.kind() {
+        "program" | "formal_parameters" | "for_statement" | "for_in_statement" | "catch_clause"
+        | "switch_body" | "with_statement" => true,
+        "statement_block" => !node

+            .is_some_and(|p| is_js_ts_function_like(p.kind())),
+        k => is_js_ts_function_like(k) || is_class(k),
+    }

+/// A node whose extent errors cannot move (SPEC E6): a braced function or a class.
+fn sealing(node: Node<'_>) -> bool {
+    // MEASUREMENT ONLY: option E6b also seals type-space containers.
+    let type_space = std::env::var_os("PRISM_S1B_E6_TYPESPACE").is_some()
+        && matches!(
+            node.kind(),
+            "interface_declaration"
+                | "type_alias_declaration"
+                | "type_parameters"
+                | "type_arguments"
+                | "type_annotation"
+        );
+    type_space
+        || is_class(node.kind())
+        || node.kind() == "class_static_block"
+        || (is_js_ts_function_like(node.kind())
+            && node
+                .child_by_field_name("body")
+                .is_some_and(|b| b.kind() == "statement_block"))

+    /// What `name`, used at `ident`, denotes (SPEC §3.1).
+    pub(crate) fn js_ts_site_binding<'a>(

+        name: &str,
+        cache: &mut JsBindingCache<'a>,
+    ) -> JsBinding {
+        let result = match self.js_ts_binding_walk(ident, name, cache) {
+            Walk::Found(scope, decls) => self.js_ts_classify(scope, &decls, name, ident, cache),
+            Walk::Unbound => JsBinding::Refused("unbound"),
+            Walk::Base(reason) => JsBinding::Base(reason),
+        };
+        if std::env::var_os("PRISM_S1B_DEBUG").is_some() {
+            eprintln!(
+                "S1B {}:{} {name} {result:?}",
+                self.path,
+                ident.start_position().row + 1
+            );
+        }
+        result
+    }

+            return Strictness::Strict;
+        }
+        let mut up = Some(scope);
+        while let Some(n) = up {
+            let prologue = match n.kind() {
+                "program" => Some(n),
+                k if is_js_ts_function_like(k) => n.child_by_field_name("body"),
+                "class_body" | "class_static_block" => return Strictness::Strict,
+                _ => None,
+            };
+            let mut cursor = prologue.map(|p| p.walk());
+            if let (Some(p), Some(c)) = (prologue, cursor.as_mut()) {
+                for st in p.named_children(c) {
+                    let Some(lit) = st
+                        .named_child(0)
+                        .filter(|l| st.kind() == "expression_statement" && l.kind() == "string")
+                    else {
+                        break;
+                    };
+                    if matches!(self.node_text(&lit), "'use strict'" | "\"use strict\"") {
+                        return Strictness::Strict;
+                    }
+                }
+            }
+            up = n.parent();
+        }
+        if self.path.ends_with(".cjs") {
+            Strictness::Sloppy
+        } else {
+            Strictness::Unknown
+        }
+    }

+    /// Parameters, the implicit `arguments` of a non-arrow function, and a function
+    /// expression's own name. `at` is the node recorded as their declaration.
+    fn js_ts_function_names<'a>(&self, f: Node<'a>, at: Node<'a>, out: &mut Index<'a>) {
+        let mut params = BTreeSet::new();
+        self.collect_js_ts_parameter_bindings(f, &mut params);
+        if f.kind() != "arrow_function" {
+            params.insert("arguments".to_string());
+        }
+        for p in params {
+            out.entry(p).or_default().push(at);
+        }
+        if matches!(f.kind(), "function_expression" | "generator_function") {
+            if let Some(n) = f.child_by_field_name("name") {
+                out.entry(self.node_text(&n).into()).or_default().push(f);
+            }
+        }
+    }

+    /// Declarations under `node` (SPEC §3.1): lexical ones while `direct` (still in the
+    /// scope's own statement list), `var` ones anywhere while `hoist`. A hoisted `var` whose
+    /// name a `catch` parameter binds, or any under `with`, records its tainting node (a
+    /// non-callable marker, Annex B.3.4). A block function reached while hoisting records
+    /// its block (Annex B.3.3). Nested functions, classes, static blocks and namespace
+    /// bodies are their own scopes.
+    fn js_ts_declare_walk<'a>(
+        &'a self,
+        node: Node<'a>,
+        (direct, hoist, strict): (bool, bool, Strictness),
+        taint: &BTreeSet<String>,
+        out: &mut Index<'a>,
+        annex: &mut BTreeSet<String>,
+    ) {
+        let mut cursor = node.walk();
+        for ch in node.named_children(&mut cursor) {
+            let kind = ch.kind();
+            let own = ch.child_by_field_name("name").map(|n| {
+                let text = self.node_text(&n);
+                text.split('.').next().unwrap_or(text).to_string()
+            });
+            let mut add = |name: String, at: Node<'a>| out.entry(name).or_default().push(at);
+            let declares_name = matches!(
+                kind,
+                "function_declaration"
+                    | "generator_function_declaration"
+                    | "class_declaration"
+                    | "abstract_class_declaration"
+                    | "internal_module"
+                    | "module"
+                    | "enum_declaration"
+            );
+            if is_js_ts_function_like(kind)
+                || is_class(kind)
+                || matches!(
+                    kind,
+                    "class_static_block" | "enum_declaration" | "decorator"
+                )
+                || matches!(kind, "internal_module" | "module")
+            {
+                // D1: Annex B.3.2 hoists only an ordinary (not async, not generator) block
+                // function declaration, and only in non-strict code.
+                let mut cursor = ch.walk();
+                let ordinary = kind == "function_declaration"
+                    && !ch.children(&mut cursor).any(|c| c.kind() == "async");
+                match own {
+                    Some(n) if direct && declares_name => add(n, ch),
+                    Some(n) if hoist && ordinary && strict == Strictness::Sloppy => add(n, node),
+                    Some(n) if hoist && ordinary && strict == Strictness::Unknown => {
+                        annex.insert(n);
+                    }
+                    _ => {}
+                }
+                continue;
+            }
+            let marker = |n: &String| taint.contains(n) || taint.contains("");
+            match kind {
+                "variable_declarator" => {
+                    let mut names = BTreeSet::new();
+                    if let Some(p) = ch.child_by_field_name("name") {
+                        self.collect_js_ts_binding_pattern_names(p, &mut names);
+                    }
+                    for n in names {
+                        let at = if marker(&n) { node } else { ch };
+                        add(n, at);
+                    }
+                    continue;
+                }
+                "variable_declaration" if !hoist => continue,
+                "lexical_declaration" if !direct => continue,
+                "interface_declaration" | "type_alias_declaration" => continue,
+                "import_statement" | "import_alias" => {
+                    if direct {
+                        for n in self.js_ts_import_value_names(ch) {
+                            add(n, ch);
+                        }
+                    }
+                    continue;
+                }
+                "function_signature" => {
+                    if let Some(n) = own.filter(|_| direct && node.kind() == "ambient_declaration")
+                    {
+                        add(n, ch);
+                    }
+                    continue;
+                }
+                "for_in_statement" if hoist => {
+                    let is_var = ch
+                        .child_by_field_name("kind")
+                        .is_some_and(|k| self.node_text(&k) == "var");
+                    let mut names = BTreeSet::new();
+                    if let Some(left) = ch.child_by_field_name("left").filter(|_| is_var) {
+                        self.collect_js_ts_binding_pattern_names(left, &mut names);
+                    }
+                    for n in names {
+                        add(n, ch);
+                    }
+                }
+                "catch_clause" => {
+                    let mut names = taint.clone();
+                    if let Some(p) = ch.child_by_field_name("parameter") {
+                        self.collect_js_ts_binding_pattern_names(p, &mut names);
+                    }
+                    if let Some(body) = ch.child_by_field_name("body") {
+                        self.js_ts_declare_walk(body, (false, hoist, strict), &names, out, annex);
+                    }
+                    continue;
+                }
+                "with_statement" => {
+                    let names = BTreeSet::from([String::new()]);
+                    if let Some(body) = ch.child_by_field_name("body") {
+                        self.js_ts_declare_walk(body, (false, hoist, strict), &names, out, annex);
+                    }
+                    continue;
+                }
+                _ => {}
+            }
+            if let Some(using) = self.js_ts_using_declaration(ch).filter(|_| direct) {
+                let mut names = BTreeSet::new();
+                if let Some(left) = using.child_by_field_name("left") {
+                    self.collect_js_ts_binding_pattern_names(left, &mut names);
+                }
+                for n in names {
+                    add(n, using);
+                }
+                continue;
+            }
+            let transparent = matches!(
+                kind,
+                "export_statement"
+                    | "expression_statement"
+                    | "labeled_statement"
+                    | "switch_case"
+                    | "switch_default"
+                    | "ambient_declaration"
+                    | "lexical_declaration"
+                    | "variable_declaration"
+            );
+            let mode = (direct && transparent, hoist, strict);
+            self.js_ts_declare_walk(ch, mode, taint, out, annex);
+        }
+    }

+    /// TS `using x = e` / `await using x = e`: the grammar spells the declaration as an
+    /// assignment expression carrying an anonymous `using` token.
+    fn js_ts_using_declaration<'a>(&self, stmt: Node<'a>) -> Option<Node<'a>> {
+        let mut expr = stmt
+            .named_child(0)
+            .filter(|_| stmt.kind() == "expression_statement")?;
+        if expr.kind() == "await_expression" {
+            expr = expr.named_child(0)?;
+        }
+        let mut cursor = expr.walk();
+        let using = expr.kind() == "assignment_expression"
+            && expr
+                .children(&mut cursor)
+                .any(|c| !c.is_named() && c.kind() == "using");
+        using.then_some(expr)
+    }

+    fn js_ts_has_function_argument(&self, call: Node<'_>) -> bool {
+        let Some(args) = call.child_by_field_name("arguments") else {
+            return false;
+        };
+        let mut cursor = args.walk();
+        let found = args
+            .named_children(&mut cursor)
+            .any(|a| matches!(a.kind(), "arrow_function" | "function_expression"));
+        found
+    }

+    /// Value names an `import` statement or `import x = …` alias binds (type-only ones bind
+    /// nothing in value space).
+    fn js_ts_import_value_names(&self, node: Node<'_>) -> Vec<String> {
+        if node.kind() == "import_alias" {
+            return node
+                .named_child(0)
+                .map(|n| self.node_text(&n).into())
+                .into_iter()
+                .collect();
+        }
+        if self.js_ts_import_statement_is_type_only(node) {
+            return Vec::new();
+        }
+        let mut out = Vec::new();
+        let mut stack = vec![node];
+        while let Some(n) = stack.pop() {
+            let mut cursor = n.walk();
+            for c in n.named_children(&mut cursor) {
+                match c.kind() {
+                    "identifier" => out.push(self.node_text(&c).into()),
+                    "import_specifier" if !self.js_ts_import_specifier_is_type_only(c) => {
+                        let local = c
+                            .child_by_field_name("alias")
+                            .or(c.child_by_field_name("name"));
+                        out.extend(local.map(|l| self.node_text(&l).to_string()));
+                    }
+                    "import_clause"
+                    | "namespace_import"
+                    | "named_imports"
+                    | "import_require_clause" => stack.push(c),
+                    _ => {}
+                }
+            }
+        }
+        out
+    }

+    /// Classification of a binding (SPEC §3.1, B0–B3 and the may-call rule M).
+    fn js_ts_classify<'a>(
+        &'a self,
+        scope: Node<'a>,
+        decls: &[Node<'a>],
+        name: &str,
+        site: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> JsBinding {
+        if std::env::var_os("PRISM_S1B_NO_FAILSAFE").is_none() {
+            if let Err(reason) = self.js_ts_scope_clean(scope, cache) {
+                return JsBinding::Refused(reason);
+            }
+        }
+        let broad = std::env::var_os("PRISM_S1B_BROAD_E6").is_some() && scope.has_error();
+        if broad || !self.js_ts_recovery_sealed(scope, name, site, cache) {
+            return JsBinding::Refused("parse_recovery");
+        }
+        if scope.kind() == "with_statement" {
+            return JsBinding::Refused("with");
+        }
+        let [decl] = decls else {
+            return JsBinding::Refused("duplicate_declaration");
+        };
+        let fe_self = matches!(decl.kind(), "function_expression" | "generator_function")
+            && decl
+                .child_by_field_name("name")
+                .is_some_and(|n| self.node_text(&n) == name);
+        let using = decl.kind() == "assignment_expression";
+        let declarator = (decl.kind() == "variable_declarator" || using)
+            && decl
+                .child_by_field_name(if using { "left" } else { "name" })
+                .is_some_and(|n| n.kind() == "identifier");
+        let function = decl.kind().ends_with("function_declaration");
+        if (fe_self || declarator || function) && self.js_ts_written(scope, name, cache) {
+            return JsBinding::MayCall;
+        }
+        let callable = if function || fe_self {
+            *decl
+        } else if declarator {
+            let Some(mut value) = decl.child_by_field_name(if using { "right" } else { "value" })
+            else {
+                return JsBinding::Refused("not_callable");
+            };
+            // B3 unwraps parentheses, TS assertions and assignment chains (`(M.f = function(){})`).
+            while let Some(inner) = match value.kind() {
+                "parenthesized_expression"
+                | "as_expression"
+                | "satisfies_expression"
+                | "non_null_expression" => value.named_child(0),
+                "type_assertion" => value.named_child(value.named_child_count().saturating_sub(1)),
+                "assignment_expression" => value.child_by_field_name("right"),
+                _ => None,
+            } {
+                value = inner;
+            }
+            match value.kind() {
+                "arrow_function" | "function_expression" => value,
+                "call_expression" if using => {
+                    return match self.js_ts_has_function_argument(value) {
+                        true => JsBinding::MayCall,
+                        false => JsBinding::Refused("not_callable"),
+                    };
+                }
+                "call_expression" => {
+                    let Some(parent) = decl.parent() else {
+                        return JsBinding::Refused("not_callable");
+                    };
+                    return match self.js_ts_wrapped_export(parent, *decl, true) {
+                        Ok((local, start_line, end_line)) => JsBinding::Callable(JsTerminal {
+                            local,
+                            start_line,
+                            end_line,
+                            wrapped: !self.js_ts_passthrough_wrapper(value),
+                        }),
+                        // E5: a Pattern-3 over-named argument keeps base behavior; a call
+                        // with no function argument holds no callable.
+                        Err(_) if self.js_ts_has_function_argument(value) => JsBinding::MayCall,
+                        Err(_) => JsBinding::Refused("not_callable"),
+                    };
+                }
+                _ => return JsBinding::Refused("not_callable"),
+            }
+        } else {
+            return JsBinding::Refused("not_callable");
+        };
+        let Some(local) = self.language.function_name(&callable) else {
+            return JsBinding::Refused("unindexed");
+        };
+        let (start_line, end_line) = self.node_line_range(&callable);
+        JsBinding::Callable(JsTerminal {
+            local: self.node_text(&local).to_string(),
+            start_line,
+            end_line,
+            wrapped: false,
+        })
+    }

+    /// B0 (fail-safe): every non-leaf named node kind in `scope` is classified, and no
+    /// identifier is spelled with an escape (the collector compares raw spellings).
+    fn js_ts_scope_clean<'a>(
+        &self,
+        scope: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> Result<(), &'static str> {
+        if let Some(r) = cache.clean.get(&scope.id()) {
+            return *r;
+        }
+        let mut result = Ok(());
+        let mut stack = vec![scope];
+        while let Some(n) = stack.pop() {
+            if n.is_named() && !n.is_error() && !n.is_missing() && !classified(n.kind()) {
+                result = Err("unclassified_kind");
+                break;
+            }
+            if matches!(
+                n.kind(),
+                "identifier" | "shorthand_property_identifier_pattern"
+            ) && self.node_text(&n).contains('\\')
+            {
+                result = Err("escaped_identifier");
+                break;
+            }
+            let mut cursor = n.walk();
+            stack.extend(n.named_children(&mut cursor));
+        }
+        cache.clean.insert(scope.id(), result);
+        result
+    }

+    /// B1 (owner E6, the narrower rule): a binding scope with a parse error is usable only
+    /// when no brace is missing or inside an error anywhere in the file (so every braced
+    /// node's extent is its true extent), and every error in the scope lies in a braced
+    /// function, class or static block that does not contain the site (whatever such a node
+    /// declares is invisible outside it).
+    fn js_ts_recovery_sealed<'a>(
+        &self,
+        scope: Node<'a>,
+        _name: &str,
+        site: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> bool {
+        if !scope.has_error() {
+            return true;
+        }
+        let root = self.tree.root_node();
+        // (i) only structural brace tokens under an error, or missing braces, break it.
+        let braces = *cache.braces.get_or_insert_with(|| {
+            fn structural(n: Node<'_>) -> bool {
+                let mut cursor = n.walk();
+                let found = n.children(&mut cursor).any(|c| {
+                    (!c.is_named() && matches!(c.kind(), "{" | "}"))
+                        || (c.is_error() && structural(c))
+                });
+                found
+            }
+            let mut ok = true;
+            let mut stack = vec![root];
+            while let Some(n) = stack.pop().filter(|_| ok) {
+                ok = !(n.is_missing() && matches!(n.kind(), "{" | "}"))
+                    && !(n.is_error() && structural(n));
+                let mut cursor = n.walk();
+                stack.extend(
+                    n.children(&mut cursor)
+                        .filter(|c| c.has_error() || c.is_missing()),
+                );
+            }
+            ok
+        });
+        let inside = |outer: Node<'_>, inner: Node<'_>| {
+            outer.start_byte() <= inner.start_byte() && inner.end_byte() <= outer.end_byte()
+        };
+        // (ii) sealed only inside a delimited child (paired braces or parentheses) of a
+        // sealing node strictly inside the scope that does not contain the site.
+        let delimited = |d: Node<'_>| {
+            let (open, close) = match d.kind() {
+                "statement_block" | "class_body" => ("{", "}"),
+                "formal_parameters" => ("(", ")"),
+                _ => return false,
+            };
+            let first = d.child(0);
+            let last = d.child(d.child_count().saturating_sub(1));
+            d.parent().is_some_and(sealing)
+                && first.is_some_and(|c| c.kind() == open && !c.is_missing())
+                && last.is_some_and(|c| c.kind() == close && !c.is_missing())
+        };
+        let mut stack = vec![scope];
+        while let Some(n) = stack.pop() {
+            if n.is_error() || n.is_missing() {
+                let mut up = n.parent();
+                while let Some(d) = up.filter(|d| d.id() != scope.id() && !delimited(*d)) {
+                    up = d.parent();
+                }
+                let sealed = up.and_then(|d| d.parent()).is_some_and(|f| {
+                    up.is_some_and(|d| d.id() != scope.id())
+                        && f.id() != scope.id()
+                        && inside(scope, f)
+                        && !inside(f, site)
+                });
+                if !braces || !sealed {
+                    return false;
+                }
+            }
+            let mut cursor = n.walk();
+            stack.extend(
+                n.children(&mut cursor)
+                    .filter(|c| c.has_error() || c.is_missing()),
+            );
+        }
+        true
+    }

+    /// B5 input: a write whose target resolves to `scope`'s binding of `name` (or passes
+    /// through a `with`). Module scope uses the base scan unless S1b-2's scoped scan is on.
+    fn js_ts_written<'a>(
+        &'a self,
+        scope: Node<'a>,
+        name: &str,
+        cache: &mut JsBindingCache<'a>,
+    ) -> bool {
+        if scope.kind() == "program" && std::env::var_os("PRISM_S1B_SCOPED_MODULE_SCAN").is_none() {
+            return self.js_ts_module_value_written(name);
+        }
+        if !cache.writes.contains_key(&scope.id()) {
+            let mut index = Index::new();
+            let mut stack = vec![scope];
+            while let Some(n) = stack.pop() {
+                let mut cursor = n.walk();
+                let using = n
+                    .children(&mut cursor)
+                    .any(|c| !c.is_named() && c.kind() == "using");
+                let write = (matches!(
+                    n.kind(),
+                    "assignment_expression"
+                        | "augmented_assignment_expression"
+                        | "update_expression"
+                ) && !using)
+                    || (n.kind() == "for_in_statement" && n.child_by_field_name("kind").is_none());
+                if let Some(target) = n
+                    .child_by_field_name("left")
+                    .or_else(|| n.child_by_field_name("argument"))
+                    .filter(|_| write)
+                {
+                    let mut names = BTreeSet::new();
+                    self.collect_js_ts_binding_pattern_names(target, &mut names);
+                    for w in names {
+                        index.entry(w).or_default().push(target);
+                    }
+                }
+                let mut cursor = n.walk();
+                stack.extend(n.named_children(&mut cursor));
+            }
+            cache.writes.insert(scope.id(), index);
+        }
+        let targets = cache.writes[&scope.id()]
+            .get(name)
+            .cloned()
+            .unwrap_or_default();
+        targets
+            .into_iter()
+            .any(|t| match self.js_ts_binding_walk(t, name, cache) {
+                Walk::Found(s, _) => s.id() == scope.id() || s.kind() == "with_statement",
+                Walk::Base(_) => true,
+                Walk::Unbound => false,
+            })
+    }

+    /// S1b-4: `None` when `q` is not one of the file's ESM namespace-import locals; else the
+    /// qualifier's binding: proven to be that import (unique, clean, unwritten), refuted, or
+    /// at an unproven position (base behavior).
+    pub(crate) fn js_ts_namespace_proven<'a>(
+        &'a self,
+        q: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> Option<JsBinding> {
+        let name = self.node_text(&q);
+        let namespaces = cache.namespaces.get_or_insert_with(|| {
+            self.extract_import_bindings()
+                .into_iter()
+                .filter(|b| b.kind == crate::call_graph::ImportBindingKind::ModuleImport)
+                .map(|b| b.local)
+                .collect()
+        });
+        if !namespaces.contains(name) {
+            return None;
+        }
+        let (scope, decls) = match self.js_ts_binding_walk(q, name, cache) {
+            Walk::Found(scope, decls) => (scope, decls),
+            Walk::Unbound => return Some(JsBinding::Refused("unbound")),
+            Walk::Base(reason) => return Some(JsBinding::Base(reason)),
+        };
+        let proven = scope.kind() == "program"
+            && matches!(decls.as_slice(), [d] if d.kind() == "import_statement")
+            && self.js_ts_scope_clean(scope, cache).is_ok()
+            && self.js_ts_recovery_sealed(scope, name, q, cache)
+            && !self.js_ts_written(scope, name, cache);
+        Some(if proven {
+            JsBinding::MayCall // prototype encoding: "proven namespace import"
+        } else {
+            JsBinding::Refused("namespace_unproven")
+        })
+    }

+    /// ECMAScript `ModuleExportName` / enum member StringValue: an identifier's text, or a
+    /// string literal's value with its escapes decoded (both quote forms).
+    pub(crate) fn js_ts_module_export_name(&self, node: Node<'_>) -> String {
+        if node.kind() != "string" {
+            return self.node_text(&node).to_string();
+        }
+        let mut out = String::new();
+        let mut cursor = node.walk();
+        for part in node.named_children(&mut cursor) {
+            let text = self.node_text(&part);
+            if part.kind() != "escape_sequence" {
+                out.push_str(text);
+                continue;
+            }
+            let body = &text[1..];
+            let hex = body
+                .trim_start_matches(['u', 'x', '{'])
+                .trim_end_matches('}');
+            let decoded = match body.chars().next() {
+                Some('n') => Some('\n'),
+                Some('t') => Some('\t'),
+                Some('r') => Some('\r'),
+                Some('u' | 'x') => u32::from_str_radix(hex, 16).ok().and_then(char::from_u32),
+                Some(c) => Some(c),
+                None => None,
+            };
+            out.extend(decoded);
+        }
+        out
+    }

+    /// D4: the target an ESM local export records (SPEC §3.2).
+    pub(super) fn js_ts_local_export_target(
