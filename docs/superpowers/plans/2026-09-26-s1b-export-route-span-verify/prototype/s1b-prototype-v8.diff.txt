+//! the name decides; the binding denotes one callable only when that scope holds exactly one
+//! declaration, the declaration's value is a function (or an admitted React wrapper of one),
+//! the scope parses cleanly (or its errors are sealed), and every node kind in the scope is
+//! classified. A written binding, or a declarator whose value is any other call, is the

+fn classified(kind: &str) -> bool {
+    static SET: OnceLock<BTreeSet<&'static str>> = OnceLock::new();
+    SET.get_or_init(|| CLASSIFIED.split_whitespace().collect())
+        .contains(kind)

+fn is_class(kind: &str) -> bool {
+    matches!(
+        kind,
+        "class_declaration" | "class" | "abstract_class_declaration"
+    )
+}

+/// Σ (SPEC §3.1): the node kinds that open an environment.
+fn is_scope(node: Node<'_>) -> bool {
+    match node.kind() {
+        "program" | "formal_parameters" | "for_statement" | "for_in_statement" | "catch_clause"
+        | "switch_body" | "with_statement" => true,
+        "statement_block" => !node
+            .parent()
+            .is_some_and(|p| is_js_ts_function_like(p.kind())),
+        k => is_js_ts_function_like(k) || is_class(k),
+    }
+}

+/// The next node up from `node` on a site's scope walk. A parameter list sees its own
+/// function's parameters but not the body, so the walk resumes above that function; a
+/// decorator is evaluated where its class is defined, outside the class.
+fn next_up(node: Node<'_>) -> Option<Node<'_>> {
+    match node.kind() {
+        "formal_parameters" => node.parent().and_then(|f| f.parent()),
+        "decorator" => {
+            let mut up = node.parent();
+            while let Some(n) = up.filter(|n| !is_class(n.kind())) {
+                up = n.parent();
+            }
+            up.and_then(|class| class.parent())
+        }
+        _ => node.parent(),
+    }
+}

+/// A node whose extent errors cannot move (SPEC E6): a braced function or a class.
+fn sealing(node: Node<'_>) -> bool {
+    // MEASUREMENT ONLY: option E6b also seals type-space containers.
+    let type_space = std::env::var_os("PRISM_S1B_E6_TYPESPACE").is_some()
+        && matches!(
+            node.kind(),
+            "interface_declaration"
+                | "type_alias_declaration"
+                | "type_parameters"
+                | "type_arguments"
+                | "type_annotation"
+        );
+    type_space
+        || is_class(node.kind())
+        || node.kind() == "class_static_block"
+        || (is_js_ts_function_like(node.kind())
+            && node
+                .child_by_field_name("body")
+                .is_some_and(|b| b.kind() == "statement_block"))
+}

+    /// What `name`, used at `ident`, denotes (SPEC §3.1).
+    pub(crate) fn js_ts_site_binding<'a>(
+        &'a self,
+        ident: Node<'a>,
+        name: &str,
+        cache: &mut JsBindingCache<'a>,
+    ) -> JsBinding {
+        let result = match self.js_ts_binding_scope(ident, name, cache) {
+            Some((scope, decls)) => self.js_ts_classify(scope, &decls, name, ident, cache),
+            None => JsBinding::Refused("unbound"),
+        };
+        if std::env::var_os("PRISM_S1B_DEBUG").is_some() {
+            eprintln!(
+                "S1B {}:{} {name} {result:?}",
+                self.path,
+                ident.start_position().row + 1
+            );
+        }
+        result
+    }

+    /// The nearest scope enclosing `site` that declares `name`, and its declarations.
+    fn js_ts_binding_scope<'a>(
+        &'a self,
+        site: Node<'a>,
+        name: &str,
+        cache: &mut JsBindingCache<'a>,
+    ) -> Option<(Node<'a>, Vec<Node<'a>>)> {
+        let mut current = site.parent();
+        while let Some(scope) = current {
+            if scope.kind() == "with_statement" {
+                return Some((scope, vec![scope]));
+            }
+            if is_scope(scope) {
+                let index = cache
+                    .decls
+                    .entry(scope.id())
+                    .or_insert_with(|| self.js_ts_scope_index(scope));
+                if let Some(decls) = index.get(name) {
+                    return Some((scope, decls.clone()));
+                }
+            }
+            current = next_up(scope);
+        }
+        None
+    }

+    /// Every name `scope` declares itself, with the declaring nodes (SPEC §3.1 table).
+    fn js_ts_scope_index<'a>(&'a self, scope: Node<'a>) -> Index<'a> {
+        let mut out = Index::new();
+        let mut names = BTreeSet::new();
+        let kind = scope.kind();
+        match kind {
+            "catch_clause" | "for_in_statement" => {
+                let lexical = kind == "catch_clause"
+                    || scope
+                        .child_by_field_name("kind")
+                        .is_some_and(|k| matches!(self.node_text(&k), "let" | "const"));
+                let field = if kind == "catch_clause" {
+                    "parameter"
+                } else {
+                    "left"
+                };
+                if let Some(p) = scope.child_by_field_name(field).filter(|_| lexical) {
+                    self.collect_js_ts_binding_pattern_names(p, &mut names);
+                }
+            }
+            "for_statement" => {
+                if let Some(init) = scope
+                    .child_by_field_name("initializer")
+                    .filter(|i| i.kind() == "lexical_declaration")
+                {
+                    self.js_ts_declare_walk(init, true, false, &BTreeSet::new(), &mut out);
+                }
+            }
+            "formal_parameters" | "statement_block" | "switch_body" | "program" => {}
+            k if is_class(k) => {
+                names.extend(
+                    scope
+                        .child_by_field_name("name")
+                        .map(|n| self.node_text(&n).into()),
+                );
+            }
+            _ => {}
+        }
+        for name in names {
+            out.entry(name).or_default().push(scope);
+        }
+        let parent = scope.parent();
+        match kind {
+            "formal_parameters" => {
+                if let Some(f) = parent.filter(|f| is_js_ts_function_like(f.kind())) {
+                    self.js_ts_function_names(f, scope, &mut out);
+                }
+            }
+            k if is_js_ts_function_like(k) => {
+                // Parameters are recorded at their list (or the single arrow parameter), which
+                // is never a callable declaration.
+                let at = scope
+                    .child_by_field_name("parameters")
+                    .or_else(|| scope.child_by_field_name("parameter"))
+                    .unwrap_or(scope);
+                self.js_ts_function_names(scope, at, &mut out);
+                if let Some(body) = scope
+                    .child_by_field_name("body")
+                    .filter(|b| b.kind() == "statement_block")
+                {
+                    self.js_ts_declare_walk(body, true, true, &BTreeSet::new(), &mut out);
+                }
+            }
+            "statement_block" => {
+                let var_scope = parent.is_some_and(|p| {
+                    matches!(
+                        p.kind(),
+                        "class_static_block" | "internal_module" | "module"
+                    )
+                });
+                self.js_ts_declare_walk(scope, true, var_scope, &BTreeSet::new(), &mut out);
+            }
+            "switch_body" | "program" => {
+                self.js_ts_declare_walk(scope, true, kind == "program", &BTreeSet::new(), &mut out);
+            }
+            _ => {}
+        }
+        out
+    }

+                    | "enum_declaration"
+            );
+            if is_js_ts_function_like(kind)
+                || is_class(kind)
+                || matches!(
+                    kind,
+                    "class_static_block" | "enum_declaration" | "decorator"
+                )
+                || matches!(kind, "internal_module" | "module")

+                match own {
+                    Some(n) if direct && declares_name => add(n, ch),
+                    Some(n) if hoist && kind.ends_with("function_declaration") => add(n, node),
+                    _ => {}
+                }
+                continue;
+            }
+            let marker = |n: &String| taint.contains(n) || taint.contains("");
+            match kind {
+                "variable_declarator" => {
+                    let mut names = BTreeSet::new();
+                    if let Some(p) = ch.child_by_field_name("name") {
+                        self.collect_js_ts_binding_pattern_names(p, &mut names);
+                    }
+                    for n in names {

+                        add(n, at);
+                    }
+                    continue;
+                }
+                "variable_declaration" if !hoist => continue,
+                "lexical_declaration" if !direct => continue,

+                "import_statement" | "import_alias" => {

+                    }
+                    continue;
+                }
+                "function_signature" => {

+                    {
+                        add(n, ch);
+                    }
+                    continue;
+                }
+                "for_in_statement" if hoist => {

+                        .child_by_field_name("kind")
+                        .is_some_and(|k| self.node_text(&k) == "var");
+                    let mut names = BTreeSet::new();
+                    if let Some(left) = ch.child_by_field_name("left").filter(|_| is_var) {
+                        self.collect_js_ts_binding_pattern_names(left, &mut names);
+                    }
+                    for n in names {
+                        add(n, ch);
+                    }
+                }
+                "catch_clause" => {
+                    let mut names = taint.clone();
+                    if let Some(p) = ch.child_by_field_name("parameter") {
+                        self.collect_js_ts_binding_pattern_names(p, &mut names);
+                    }
+                    if let Some(body) = ch.child_by_field_name("body") {
+                        self.js_ts_declare_walk(body, false, hoist, &names, out);
+                    }
+                    continue;
+                }
+                "with_statement" => {
+                    let names = BTreeSet::from([String::new()]);
+                    if let Some(body) = ch.child_by_field_name("body") {
+                        self.js_ts_declare_walk(body, false, hoist, &names, out);
+                    }
+                    continue;
+                }
+                _ => {}
+            }
+            if let Some(using) = self.js_ts_using_declaration(ch).filter(|_| direct) {
+                let mut names = BTreeSet::new();
+                if let Some(left) = using.child_by_field_name("left") {
+                    self.collect_js_ts_binding_pattern_names(left, &mut names);
+                }
+                for n in names {
+                    add(n, using);
+                }
+                continue;
+            }
+            let transparent = matches!(
+                kind,
+                "export_statement"
+                    | "expression_statement"
+                    | "labeled_statement"
+                    | "switch_case"
+                    | "switch_default"
+                    | "ambient_declaration"
+                    | "lexical_declaration"
+                    | "variable_declaration"
+            );
+            self.js_ts_declare_walk(ch, direct && transparent, hoist, taint, out);
+        }

+    /// TS `using x = e` / `await using x = e`: the grammar spells the declaration as an
+    /// assignment expression carrying an anonymous `using` token.
+    fn js_ts_using_declaration<'a>(&self, stmt: Node<'a>) -> Option<Node<'a>> {
+        let mut expr = stmt
+            .named_child(0)
+            .filter(|_| stmt.kind() == "expression_statement")?;

+            && expr
+                .children(&mut cursor)
+                .any(|c| !c.is_named() && c.kind() == "using");
+        using.then_some(expr)

+    /// Value names an `import` statement or `import x = …` alias binds (type-only ones bind
+    /// nothing in value space).
+    fn js_ts_import_value_names(&self, node: Node<'_>) -> Vec<String> {

+    /// Classification of a binding (SPEC §3.1, B0–B3 and the may-call rule M).
+    fn js_ts_classify<'a>(
+        &'a self,
+        scope: Node<'a>,
+        decls: &[Node<'a>],
+        name: &str,
+        site: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> JsBinding {
+        if std::env::var_os("PRISM_S1B_NO_FAILSAFE").is_none() {
+            if let Err(reason) = self.js_ts_scope_clean(scope, cache) {
+                return JsBinding::Refused(reason);
+            }
+        }
+        let broad = std::env::var_os("PRISM_S1B_BROAD_E6").is_some() && scope.has_error();
+        if broad || !self.js_ts_recovery_sealed(scope, name, site, cache) {
+            return JsBinding::Refused("parse_recovery");
+        }
+        if scope.kind() == "with_statement" {
+            return JsBinding::Refused("with");
+        }
+        let [decl] = decls else {
+            return JsBinding::Refused("duplicate_declaration");
+        };
+        let fe_self = matches!(decl.kind(), "function_expression" | "generator_function")
+            && decl
+                .child_by_field_name("name")
+                .is_some_and(|n| self.node_text(&n) == name);
+        let using = decl.kind() == "assignment_expression";
+        let declarator = (decl.kind() == "variable_declarator" || using)
+            && decl
+                .child_by_field_name(if using { "left" } else { "name" })
+                .is_some_and(|n| n.kind() == "identifier");
+        let function = decl.kind().ends_with("function_declaration");
+        if (fe_self || declarator || function) && self.js_ts_written(scope, name, cache) {
+            return JsBinding::MayCall;
+        }
+        let callable = if function || fe_self {
+            *decl
+        } else if declarator {
+            let Some(mut value) = decl.child_by_field_name(if using { "right" } else { "value" })
+            else {
+                return JsBinding::Refused("not_callable");
+            };
+            // B3 unwraps parentheses, TS assertions and assignment chains (`(M.f = function(){})`).
+            while let Some(inner) = match value.kind() {
+                "parenthesized_expression"
+                | "as_expression"
+                | "satisfies_expression"
+                | "non_null_expression" => value.named_child(0),
+                "type_assertion" => value.named_child(value.named_child_count().saturating_sub(1)),
+                "assignment_expression" => value.child_by_field_name("right"),
+                _ => None,
+            } {
+                value = inner;
+            }
+            match value.kind() {
+                "arrow_function" | "function_expression" => value,
+                "call_expression" if using => {
+                    return match self.js_ts_has_function_argument(value) {
+                        true => JsBinding::MayCall,
+                        false => JsBinding::Refused("not_callable"),
+                    };
+                }
+                "call_expression" => {
+                    let Some(parent) = decl.parent() else {
+                        return JsBinding::Refused("not_callable");
+                    };
+                    return match self.js_ts_wrapped_export(parent, *decl, true) {
+                        Ok((local, start_line, end_line)) => JsBinding::Callable(JsTerminal {
+                            local,
+                            start_line,
+                            end_line,
+                            wrapped: !self.js_ts_passthrough_wrapper(value),
+                        }),
+                        // E5: a Pattern-3 over-named argument keeps base behavior; a call
+                        // with no function argument holds no callable.
+                        Err(_) if self.js_ts_has_function_argument(value) => JsBinding::MayCall,
+                        Err(_) => JsBinding::Refused("not_callable"),
+                    };
+                }
+                _ => return JsBinding::Refused("not_callable"),
+            }
+        } else {
+            return JsBinding::Refused("not_callable");
+        };
+        let Some(local) = self.language.function_name(&callable) else {
+            return JsBinding::Refused("unindexed");
+        };
+        let (start_line, end_line) = self.node_line_range(&callable);
+        JsBinding::Callable(JsTerminal {
+            local: self.node_text(&local).to_string(),
+            start_line,
+            end_line,
+            wrapped: false,
+        })
+    }

+    /// B0 (fail-safe): every non-leaf named node kind in `scope` is classified, and no
+    /// identifier is spelled with an escape (the collector compares raw spellings).
+    fn js_ts_scope_clean<'a>(
+        &self,
+        scope: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> Result<(), &'static str> {
+        if let Some(r) = cache.clean.get(&scope.id()) {
+            return *r;
+        }
+        let mut result = Ok(());
+        let mut stack = vec![scope];
+        while let Some(n) = stack.pop() {
+            if n.named_child_count() > 0 && !n.is_error() && !classified(n.kind()) {
+                result = Err("unclassified_kind");
+                break;
+            }
+            if matches!(
+                n.kind(),
+                "identifier" | "shorthand_property_identifier_pattern"
+            ) && self.node_text(&n).contains('\\')
+            {
+                result = Err("escaped_identifier");
+                break;
+            }
+            let mut cursor = n.walk();
+            stack.extend(n.named_children(&mut cursor));
+        }
+        cache.clean.insert(scope.id(), result);
+        result
+    }

+    /// B1 (owner E6, the narrower rule): a binding scope with a parse error is usable only
+    /// when no brace is missing or inside an error anywhere in the file (so every braced
+    /// node's extent is its true extent), and every error in the scope lies in a braced
+    /// function, class or static block that does not contain the site (whatever such a node
+    /// declares is invisible outside it).
+    fn js_ts_recovery_sealed<'a>(
+        &self,
+        scope: Node<'a>,
+        _name: &str,
+        site: Node<'a>,
+        cache: &mut JsBindingCache<'a>,
+    ) -> bool {
+        if !scope.has_error() {
+            return true;
+        }
+        let root = self.tree.root_node();
+        let braces = *cache.braces.get_or_insert_with(|| {
+            let mut ok = true;
+            let mut stack = vec![root];
+            while let Some(n) = stack.pop().filter(|_| ok) {
+                ok = !(n.is_missing() && matches!(n.kind(), "{" | "}"))
+                    && !(n.is_error() && self.node_text(&n).contains(['{', '}']));
+                let mut cursor = n.walk();
+                stack.extend(
+                    n.children(&mut cursor)
+                        .filter(|c| c.has_error() || c.is_missing()),
+                );
+            }
+            ok
+        });
+        let inside = |outer: Node<'_>, inner: Node<'_>| {
+            outer.start_byte() <= inner.start_byte() && inner.end_byte() <= outer.end_byte()
+        };
+        let mut stack = vec![scope];
+        while let Some(n) = stack.pop() {
+            if n.is_error() || n.is_missing() {
+                let mut up = n.parent();
+                while let Some(f) = up.filter(|f| f.id() != scope.id() && !sealing(*f)) {
+                    up = f.parent();
+                }
+                let sealed = up.is_some_and(|f| f.id() != scope.id() && !inside(f, site));
+                if !braces || !sealed {
+                    return false;
+                }
+            }
+            let mut cursor = n.walk();
+            stack.extend(
+                n.children(&mut cursor)
+                    .filter(|c| c.has_error() || c.is_missing()),
+            );
+        }
+        true
+    }

+    /// through a `with`). Module scope uses the base scan unless S1b-2's scoped scan is on.

+        &'a self,
+        scope: Node<'a>,

+        cache: &mut JsBindingCache<'a>,

+            return self.js_ts_module_value_written(name);

+    }
